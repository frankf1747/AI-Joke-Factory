/* ---- interactive bits ---- */
function PasteSplitDemo() {
  const [v, setV] = useState('1) Our standup could’ve been an email.\n2) I hugged my pivot table.\n3) The AI sent a 40-slide deck.');
  const numbered = /\d{1,2}[.)]\s+/.test(v.trim());
  const lines = (numbered ? v.trim().split(/\s*\d{1,2}[.)]\s+/) : v.split('\n')).map((l) => l.trim()).filter(Boolean);
  return (
    <div>
      <textarea value={v} onChange={(e) => setV(e.target.value)} rows={3}
        className="w-full text-sm border border-gray-300 rounded-lg px-3 py-2 outline-none focus:ring-2 focus:ring-blue-400 resize-none" />
      <div className="flex items-center gap-2 my-2 text-gray-400 text-xs"><Icon name="ArrowDown" size={13} /> splits into {lines.length} jokes</div>
      <div className="space-y-1.5">
        {lines.map((l, i) => (
          <div key={i} className="bg-white border border-blue-100 border-l-[3px] border-l-blue-400 rounded-md px-2.5 py-1.5 text-sm text-gray-700">{l}</div>
        ))}
      </div>
    </div>
  );
}

function RankSubmitDemo() {
  const [cards, setCards] = useState([
    { id: 1, t: '“I scheduled a meeting to discuss having fewer meetings. Attendance was full.”' },
    { id: 2, t: '“Coffee is renewable: I renew my entire personality every cup.”' }]);
  const [selected, setSelected] = useState(() => new Set());
  const [dragIdx, setDragIdx] = useState(null);
  const move = (i, d) => setCards((c) => { const n = [...c], j = i + d; if (j < 0 || j >= n.length) return c; [n[i], n[j]] = [n[j], n[i]]; return n; });
  const endDrag = () => setDragIdx(null);
  const onDragOver = (e, i) => { e.preventDefault(); if (dragIdx === null || dragIdx === i) return; setCards((c) => { const n = [...c]; const [m] = n.splice(dragIdx, 1); n.splice(i, 0, m); return n; }); setDragIdx(i); };
  const toggle = (id) => setSelected((s) => { const n = new Set(s); n.has(id) ? n.delete(id) : n.add(id); return n; });
  return (
    <div className="space-y-2">
      {cards.map((c, i) => {
        const sel = selected.has(c.id);
        const isDefault = !selected.size && i === 0;
        return (
          <div key={c.id}
            onClick={(e) => { if (e.target.closest('button')) return; toggle(c.id); }}
            draggable
            onDragStart={(e) => { if (e.target.closest('button')) { e.preventDefault(); return; } setDragIdx(i); }}
            onDragOver={(e) => onDragOver(e, i)} onDragEnd={endDrag} onDrop={endDrag}
            className={`rounded-lg border p-3 cursor-pointer transition-colors ${dragIdx === i ? 'opacity-40' : ''} ${sel ? 'bg-green-50 border-green-400 ring-1 ring-green-400' : isDefault ? 'bg-amber-50/60 border-amber-300 border-dashed' : 'bg-white border-gray-200 hover:border-gray-300'}`}>
            <div className="flex items-start gap-2">
              <span className="shrink-0 flex flex-col items-center justify-center w-8 h-8 rounded-md bg-gray-900 text-white leading-none">
                <span className="text-[6px] font-bold uppercase tracking-wider text-gray-400">Rank</span>
                <span className="text-xs font-extrabold tabular-nums">{i + 1}</span>
              </span>
              <p className="joke-font flex-1 text-[13px] text-gray-800 leading-snug">{c.t}</p>
              <div className="shrink-0 flex flex-col items-center gap-0.5">
                <Icon name="GripVertical" size={13} className="text-gray-300" />
                <button onClick={() => move(i, -1)} disabled={i === 0} className="text-gray-400 hover:text-gray-700 disabled:opacity-25"><Icon name="ChevronUp" size={13} /></button>
                <button onClick={() => move(i, 1)} disabled={i === cards.length - 1} className="text-gray-400 hover:text-gray-700 disabled:opacity-25"><Icon name="ChevronDown" size={13} /></button>
              </div>
            </div>
            <div className="mt-2 text-[10px] font-semibold flex items-center gap-1" style={{ color: sel ? '#16a34a' : isDefault ? '#b45309' : '#94a3b8' }}>
              <Icon name={sel ? 'Check' : isDefault ? 'Star' : 'MousePointerClick'} size={11} />
              {sel ? 'Selected to submit' : isDefault ? 'Top rank — submits by default' : 'Click to submit'}
            </div>
          </div>);
      })}
      <div className="text-[11px] text-gray-400 flex items-center gap-1.5"><Icon name="GripVertical" size={12} /> Drag a card or use the arrows to rank · click a card to submit it.</div>
    </div>);
}

function CustomerDemo() {
  const [bought, setBought] = useState(false);
  useEffect(() => { const t = setInterval(() => setBought((b) => !b), 1800); return () => clearInterval(t); }, []);
  return (
    <div className="bg-gray-50 border border-gray-200 rounded-lg p-3">
      <div className="grid gap-1 mb-2" style={{ gridTemplateColumns: 'repeat(12,1fr)' }}>
        {Array.from({ length: 36 }).map((_, i) => (
          <div key={i} className="rounded-full aspect-square" style={{ background: bought && i % 3 === 0 ? BRAND.sold : '#e2e8f0', transition: 'background .4s' }} />
        ))}
      </div>
      <div className="text-[11px] text-gray-500">Perceived fit ≥ τ=7 → buy. Every {ECON.tickSeconds}s each customer re-evaluates the market.</div>
    </div>
  );
}

/* ---- main tutorial component ---- */
function Tutorial({ initialRole = 'joke_maker', onLaunch }) {
  const STEPS = useMemo(buildSteps, []);
  const [role, setRole] = useState(initialRole);
  const [step, setStep] = useState(0);
  const steps = STEPS[role];
  const meta = ROLE_META[role];
  const last = step === steps.length - 1;

  const go = (r) => { setRole(r); setStep(0); };

  return (
    <div className="max-w-5xl mx-auto">
      {/* role tabs */}
      <div className="flex flex-wrap gap-2 mb-5">
        {Object.entries(ROLE_META).map(([k, m]) => (
          <button key={k} onClick={() => go(k)}
            className={`inline-flex items-center gap-2 px-3.5 py-2 rounded-lg border text-sm font-semibold transition-colors ${role === k ? 'text-white border-transparent' : 'bg-white text-gray-600 border-gray-200 hover:border-gray-300'}`}
            style={role === k ? { background: m.color } : undefined}>
            <Icon name={m.icon} size={15} /> {m.label}
          </button>
        ))}
      </div>

      <Card className="overflow-hidden">
        <div className="grid md:grid-cols-2 gap-0 -m-4">
          {/* left: copy + steps */}
          <div className="p-6 md:p-8 flex flex-col">
            <div className="flex items-center gap-2 mb-1">
              <Badge tone="gray" icon={meta.icon}>{meta.label}</Badge>
              <span className="text-[11px] text-gray-400">{meta.tag}</span>
            </div>
            <div className="text-[11px] font-bold uppercase tracking-wider text-gray-300 mt-4 mb-1">Step {step + 1} of {steps.length}</div>
            <h2 className="text-2xl font-bold text-gray-900 leading-tight" style={{ textWrap: 'balance' }}>{steps[step].t}</h2>
            <p className="text-[15px] text-gray-600 mt-3 leading-relaxed flex-1">{steps[step].b}</p>

            {/* progress + nav */}
            <div className="mt-6">
              <div className="flex gap-1.5 mb-4">
                {steps.map((_, i) => (
                  <button key={i} onClick={() => setStep(i)} className="h-1.5 rounded-full transition-all"
                    style={{ width: i === step ? 28 : 14, background: i <= step ? meta.color : '#e5e7eb' }} />
                ))}
              </div>
              <div className="flex items-center gap-2">
                <Button variant="ghost" size="sm" onClick={() => setStep((s) => Math.max(0, s - 1))} disabled={step === 0}>
                  <Icon name="ArrowLeft" size={14} /> Back
                </Button>
                {!last ? (
                  <Button size="sm" onClick={() => setStep((s) => s + 1)} className="ml-auto">Next <Icon name="ArrowRight" size={14} /></Button>
                ) : (
                  <Button size="sm" onClick={() => onLaunch && onLaunch(role)} className="ml-auto"
                    style={{ background: meta.color }}>
                    Open the {meta.label} screen <Icon name="ArrowUpRight" size={14} />
                  </Button>
                )}
              </div>
            </div>
          </div>

          {/* right: live visual */}
          <div className="p-6 md:p-8 bg-slate-50 border-l border-gray-100 flex items-center">
            <div className="w-full">{steps[step].v()}</div>
          </div>
        </div>
      </Card>
    </div>
  );
}

Object.assign(window, { Tutorial, ROLE_META });
