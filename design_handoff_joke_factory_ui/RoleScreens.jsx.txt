/* ============================ MARKETING ============================ */
/* deterministic 0..1 from an integer — keeps the revealed dim subset stable per joke */
function mkSeed(n) {const x = Math.sin(n * 99.13) * 10000;return x - Math.floor(x);}
function ordinal(n) {const s = ['th', 'st', 'nd', 'rd'],v = n % 100;return n + (s[(v - 20) % 10] || s[v] || s[0]);}
/* dimensions with no inherent order — shown as a match/off check, not a scale */
const CATEGORICAL_DIMS = new Set(['topic', 'humor_style', 'structure']);
/* how close a joke's level on a dimension sits to the hidden ideal (1 = spot on) */
function dimProx(dim, level) {
  const lv = dim.levels.indexOf(level),il = dim.levels.indexOf(IDEAL_PROFILE[dim.id]);
  if (lv < 0 || il < 0) return 0;
  return 1 - Math.abs(lv - il) / Math.max(1, dim.levels.length - 1);
}
/* the dims a sale reveals: one of the top-3 closest + two others (seeded-random, stable) */
function revealedDims(joke) {
  const scored = DIMENSIONS.map((d) => ({ dim: d, level: joke.dims[d.id], prox: dimProx(d, joke.dims[d.id]) }));
  const ranked = [...scored].sort((a, b) => b.prox - a.prox);
  const top = ranked[Math.floor(mkSeed(joke.id) * 3) % 3];
  const rest = scored.filter((s) => s.dim.id !== top.dim.id);
  const r1 = rest[Math.floor(mkSeed(joke.id + 1) * rest.length)];
  let r2 = rest[Math.floor(mkSeed(joke.id + 7) * rest.length)];
  if (r2.dim.id === r1.dim.id) r2 = rest[(rest.indexOf(r1) + 1) % rest.length];
  return [top, r1, r2];
}

function DimScale({ dim, level, prox }) {
  const onTarget = prox >= 0.999;
  // categorical dimension → simple match / off check (no meaningful axis)
  if (CATEGORICAL_DIMS.has(dim.id)) {
    return (
      <div className="flex items-center gap-2">
        <span className="text-[10px] font-semibold text-gray-500 w-12 shrink-0">{dim.label}</span>
        <span className={`inline-flex items-center gap-1 text-[10px] font-semibold px-1.5 py-0.5 rounded ${onTarget ? 'bg-emerald-100 text-emerald-700' : 'bg-gray-100 text-gray-500'}`}>
          <Icon name={onTarget ? 'Check' : 'X'} size={11} /> {level}
        </span>
      </div>);
  }
  // scalar dimension → gray→green proximity scale on its own line (avoids label overlap)
  const frac = Math.max(0, Math.min(1, prox));
  const maxLevel = dim.levels[dim.levels.length - 1];
  return (
    <div>
      <div className="flex items-center justify-between gap-2 mb-0.5">
        <span className="text-[10px] font-semibold text-gray-500 truncate">{dim.label}</span>
        <span className={`text-[10px] truncate ${onTarget ? 'text-emerald-700 font-semibold' : 'text-gray-500'}`}>{level}</span>
      </div>
      <div className="flex items-center gap-1.5">
        <span className="text-[9px] text-gray-400 truncate max-w-[58px]">{dim.levels[0]}</span>
        <div className="relative flex-1 h-1.5 rounded-full" style={{ background: 'linear-gradient(90deg,#e5e7eb 0%,#d9f2e1 55%,#22c55e 100%)' }}>
          <span className="absolute top-1/2 rounded-full bg-white" style={{ left: `calc(5px + (100% - 10px) * ${frac})`, width: 11, height: 11, transform: 'translate(-50%,-50%)', border: `2px solid ${onTarget ? '#16a34a' : '#94a3b8'}`, boxShadow: '0 1px 2px rgb(0 0 0 / .15)' }} />
        </div>
        <span className="text-[9px] text-gray-400 truncate max-w-[58px] text-right">{maxLevel}</span>
      </div>
    </div>);
}

function SoldSignal({ entry }) {
  const dims = entry.dims;
  const top2 = [...dims].sort((a, b) => b.prox - a.prox).slice(0, 2).map((s) => s.dim.label.toLowerCase());
  return (
    <div className="rounded-lg border border-emerald-100 bg-emerald-50/40 p-3" style={{ animation: 'mkFade .4s ease both' }}>
      <div className="flex items-center gap-1.5 mb-1.5">
        <Icon name="BadgeCheck" size={13} className="text-emerald-600 shrink-0" />
        <span className="text-[11px] font-semibold text-emerald-700">Sold{entry.title ? ` · “${entry.title}”` : ''}</span>
      </div>
      <p className="text-[11px] text-gray-500 italic mb-2 line-clamp-2">{entry.text}</p>
      <p className="text-[11px] text-gray-700 mb-2.5">Bought because the <b>{top2[0]}</b> and <b>{top2[1]}</b> were right.</p>
      <div className="space-y-1.5" data-comment-anchor="bd04d5fe77-div-329-7">
        {dims.map((s) => <DimScale key={s.dim.id} dim={s.dim} level={s.level} prox={s.prox} />)}
      </div>
    </div>);
}

  const seed = JOKES[0];
  const [sold, setSold] = useState(() => [{ id: seed.id, text: seed.text, title: seed.title, dims: revealedDims(seed) }]);
  const [toast, setToast] = useState(null);

    const n = [...o],j = idx + dir;
    if (j < 0 || j >= n.length) return o;
    [n[idx], n[j]] = [n[j], n[idx]];
    return n;
  });
  const endDrag = () => setDragIdx(null);
  const onDragOver = (e, i) => {
    e.preventDefault();
    if (dragIdx === null || dragIdx === i) return;

      map((j) => ({ id: j.id, text: j.text, title: (titles[j.id] || '').trim(), dims: revealedDims(j) }));
      return [...add, ...prev];

    setTimeout(() => setToast(null), 2800);
  };

    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
      {/* left: marketing desk */}
      <div className="lg:col-span-2 space-y-6">
        <Card title="Marketing Desk" subtitle="Rank the cards (best on top) · assign Topic · click to submit · title each · release" accent={BRAND.marketing}>

            {order.map((j, idx) => {

              return (
                <div key={j.id}
                onClick={(e) => {if (e.target.closest('button, input, textarea')) return;toggle(j.id);}}
                draggable
                onDragStart={(e) => {if (e.target.closest('button, input, textarea')) {e.preventDefault();return;}setDragIdx(idx);}}

                className={`rounded-lg border p-4 cursor-pointer transition-colors ${dragging ? 'opacity-40' : ''} ${selected ? 'bg-green-50 border-green-400 ring-1 ring-green-400' : isDefault ? 'bg-amber-50/60 border-amber-300 border-dashed' : 'bg-gray-50 border-gray-200 hover:border-gray-300'}`}>
                  {/* submit status — only on cards being submitted; the whole card is clickable */}
                  {submitting &&
                  <div className={`mb-3 flex items-center justify-center gap-1.5 text-xs font-semibold rounded-md px-3 py-2 ${selected ? 'bg-green-600 text-white' : 'bg-amber-100 text-amber-800'}`}>
                    <Icon name={selected ? 'Check' : 'Star'} size={14} />

                  {/* rank badge · styled joke · drag + reorder controls (top-right) */}
                  <div className="flex items-start gap-3">
                    <span className="shrink-0 flex flex-col items-center justify-center w-11 h-11 rounded-lg bg-gray-900 text-white leading-none">
                      <span className="text-[8px] font-bold uppercase tracking-wider text-gray-400">Rank</span>
                      <span className="text-base font-extrabold tabular-nums mt-0.5">{ordinal(idx + 1)}</span>
                    </span>
                    <blockquote className="flex-1 border-l-[3px] border-violet-300 pl-3 py-0.5">
                      <p className="joke-font text-gray-900 text-[17px] leading-relaxed text-pretty">{j.text}</p>
                    </blockquote>
                    <div className="shrink-0 flex flex-col items-center gap-0.5">
                      <Icon name="GripVertical" size={16} className="text-gray-300" title="Drag the card to rank" />
                      <button onClick={() => move(idx, -1)} disabled={idx === 0} title="Rank higher"
                      className="text-gray-400 hover:text-gray-700 disabled:opacity-25 disabled:cursor-not-allowed"><Icon name="ChevronUp" size={15} /></button>
                      <button onClick={() => move(idx, 1)} disabled={idx === order.length - 1} title="Rank lower"
                      className="text-gray-400 hover:text-gray-700 disabled:opacity-25 disabled:cursor-not-allowed"><Icon name="ChevronDown" size={15} /></button>
                    </div>
                  </div>

                  <div className="mt-3">
                    <SectionLabel className="mb-1.5">Topic {submitting && <span className="text-rose-500">· required</span>}</SectionLabel>
                    <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-1.5">
                      {CATEGORIES.map((c) => {
                        const on = topics[j.id] === c.id;
                        return (
                          <button key={c.id} onClick={() => setTopics((t) => ({ ...t, [j.id]: c.id }))}
                          className={`inline-flex items-center justify-center gap-1 text-[11px] px-2 py-1.5 rounded-md border transition-colors ${on ? 'bg-violet-600 text-white border-violet-600' : 'bg-white text-gray-600 border-gray-300 hover:bg-gray-100'}`}>
                            <Icon name={c.icon} size={12} /> {c.label}
                          </button>);
                      })}
                    </div>
                  </div>

                  {/* per-joke market title — only when this joke is being submitted */}
                  {submitting &&
                  <div className="mt-3">
                    <SectionLabel className="mb-1.5">Market title <span className="text-gray-400">(this joke)</span></SectionLabel>
                    <input value={titles[j.id] || ''} onChange={(e) => setTitles((t) => ({ ...t, [j.id]: e.target.value }))}
                    placeholder={`e.g. “${j.title}”`} maxLength={120}
                    className="w-full bg-white border border-gray-300 rounded-md px-3 py-2 text-sm outline-none focus:ring-2 focus:ring-violet-300" />
                  </div>}
                </div>);
            })}

            {/* batch-level feedback — sent back to the Joke Maker */}
            <div className="rounded-lg border border-gray-200 bg-white p-4" data-comment-anchor="8663e692e6-div-343-13">
              <SectionLabel className="mb-1.5">Batch feedback <span className="text-gray-400">(sent to Joke Maker)</span></SectionLabel>
              <textarea value={batchFeedback} onChange={(e) => setBatchFeedback(e.target.value)} rows={3}
              placeholder="Notes on the whole batch — what landed, what to drop next round…"
              className="w-full bg-white border border-gray-300 rounded-md px-3 py-2 text-sm outline-none focus:ring-2 focus:ring-violet-300 resize-none placeholder-gray-400" />
            </div>

            <div className="flex items-center justify-between pt-1">
              <span className="text-[11px] text-gray-500 flex items-center gap-1.5">
                <Icon name="Zap" size={13} className="text-violet-500" />
                Drag or use ▲▼ to rank best-to-worst. Top joke submits by default — select others to add. Topic + title each, then release.
              </span>
              <Button variant="purple" onClick={release} disabled={!canRelease}>
                Release to market <Icon name="Send" size={15} />
              </Button>
            </div>
          </div>
        </Card>
      </div>

      {/* right: stats + sold-signal learning panel */}
      <div className="space-y-5">
        <div className="grid grid-cols-2 gap-3">
          <StatBox label="Current Rank" value="1" tone="rank" />
          <StatBox label="Queue" value={order.length} tone="violet" />
          <StatBox label="Avg Score" value="3.9" tone="indigo" />
          <StatBox label="Total Sales" value={sold.length} tone="amber" />
        </div>
        <Card title="What's selling — and why" subtitle="Each sale is free market research: which attributes matched customer demand" accent={BRAND.sold}>
          <div className="space-y-3 max-h-[460px] overflow-y-auto pr-1">
            {sold.length === 0 ?
            <p className="text-sm text-gray-400 italic py-4 text-center">No sales yet — release jokes to start uncovering the target.</p> :
            sold.map((e) => <SoldSignal key={e.id} entry={e} />)}
          </div>
        </Card>
      </div>

      {toast &&
      <div className="fixed bottom-6 left-1/2 -translate-x-1/2 z-50 bg-gray-900 text-white text-sm px-4 py-2.5 rounded-lg shadow-xl flex items-center gap-2">
          <Icon name="Zap" size={16} className="text-violet-400" /> {toast}
        </div>
      }
    </div>);

}
